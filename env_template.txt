# IMPORTANT (per current Supabase docs): the "direct connection" host
# (db.[PROJECT_REF].supabase.co:5432) is IPv6-only unless you buy the IPv4
# add-on. Most deploy targets (Render, Railway, Vercel) are IPv4-only, so
# use the SHARED POOLER (session mode) connection string instead — find it
# in your Supabase dashboard under Connect > Session pooler:
SUPABASE_DB_URL=postgresql://postgres.[YOUR_PROJECT_REF]:[YOUR_PASSWORD]@aws-[REGION].pooler.supabase.com:5432/postgres

# Least-privilege connection for the natural-language query path (/query).
# That path lets a language model write its own SQL, so it must not use the
# owner connection above. Create the role with sql/create_nlp_role.sql, then
# put its pooler URL here. If unset, /query falls back to SUPABASE_DB_URL,
# which works but is privileged — set this in any real deployment.
NLP_DB_URL=postgresql://nlp_reader.[YOUR_PROJECT_REF]:[NLP_PASSWORD]@aws-[REGION].pooler.supabase.com:6543/postgres
